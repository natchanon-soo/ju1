function DirectoryItem({ item }) {

    const hideTimer = useRef(null);

    const [showTooltip, setShowTooltip] = useState(false);


    const [tooltip, setTooltip] = useState({

        visible: false,
        x: 0,
        y: 0

    });

    const handleMouseEnter = (e) => {

    if (!descriptions[item.code]) return;

    // ถ้ามีการนับเวลาปิดอยู่ ยกเลิกก่อน
    clearTimeout(hideTimer.current);

    const rect = e.currentTarget.getBoundingClientRect();

    setTooltip({

        visible: true,

        x: rect.left - 370,

        y: rect.top - 60 + rect.height / 2

    });

};

    const handleMouseLeave = () => {

    hideTimer.current = setTimeout(() => {

        setTooltip(prev => ({
            ...prev,
            visible: false
        }));

    }, 20);

};

const handleTooltipEnter = () => {

    clearTimeout(hideTimer.current);

};

const handleTooltipLeave = () => {

    setTooltip(prev => ({
        ...prev,
        visible: false
    }));

};

    return (

        <>

            <div

        className={

            `dm-directory-item
            ${item.checked ? "active" : "inactive"}
            `

        }

        onMouseEnter={handleMouseEnter}

        onMouseLeave={handleMouseLeave}

    >

                <strong>

                    {item.checked ? "☑" : "☐"} {item.code}

                </strong>

                <p>{item.name}</p>

            </div>

            {

                tooltip.visible &&

                createPortal(

                    <div
    className="dm-tooltip-portal"

    onMouseEnter={handleTooltipEnter}

    onMouseLeave={handleTooltipLeave}

    style={{

        left: tooltip.x,

        top: tooltip.y

    }}
>

                        {descriptions[item.code]}

                    </div>,

                    document.body

                )

            }

        </>

    );

}






function DirectoryBox({

    title,

    items=[],

    children,

    grid="grid-2",

    icon

}){

    return(

        <div className="dm-directory-box">

            <h4>{icon} {title}</h4>

            {

                children ?

                children

                :

                <div className={`dm-directory-grid ${grid}`}>

                    {

                        items.map((item,index)=>(

                            <DirectoryItem
                                key={index}
                                item={item}
                            />

                        ))

                    }

                </div>

            }

        </div>

    )

}
